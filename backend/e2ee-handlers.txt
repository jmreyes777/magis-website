// ─────────────────────────────────────────────────────────────────--
    // E2EE Key Exchange - Route guest keys to admins and branch keys back to guests
    // ─────────────────────────────────────────────────────────────────--
    socket.on("e2ee:guest_key", async (data) => {
      console.log(`[E2EE] 🔑 Guest key received:`, { guestId: data.guestId, branchEmail: data.branchEmail });
      
      try {
        const { guestId, branchEmail, publicKey, role } = data;
        
        if (!guestId || !branchEmail || !publicKey) {
          console.warn(`[E2EE] ❌ Invalid guest key data:`, data);
          return;
        }

        // Route to all admins for this branch
        io.to(getRoomName("admin", branchEmail)).emit("e2ee:guest_key", {
          guestId,
          branchEmail,
          publicKey,
          role
        });
        
        console.log(`[E2EE] 📡 Routed guest key to admins for ${branchEmail}`);
      } catch (error) {
        console.error(`[E2EE] Error handling guest key:`, error);
      }
    });

    socket.on("e2ee:branch_key", async (data) => {
      console.log(`[E2EE] 🔑 Branch key received:`, { guestId: data.guestId, branchEmail: data.branchEmail });
      
      try {
        const { guestId, branchEmail, publicKey, role } = data;
        
        if (!guestId || !branchEmail || !publicKey) {
          console.warn(`[E2EE] ❌ Invalid branch key data:`, data);
          return;
        }

        // Route to specific guest
        io.to(getRoomName("guest", guestId)).emit("e2ee:branch_key", {
          guestId,
          branchEmail,
          publicKey,
          role
        });
        
        console.log(`[E2EE] 📡 Routed branch key to guest ${guestId}`);
      } catch (error) {
        console.error(`[E2EE] Error handling branch key:`, error);
      }
    });
